#' Compute the next value in happy number
#' 
#' @param x An positive integer.
#' 
#' @return An positive integer which is the next happy number of x
#' 
#' @examples
#' nextHappy(19)
#' nextHappy(82)

nextHappy <- function(x){
  if(length(x) != 1 || !is.numeric(x) || is.na(x) || !is.finite(x) || x <= 0 || x != floor(x)){ # make sure x is positive integer and not NA
    stop('Input must be a positive integer.')
  }
  digits <- as.integer(strsplit(as.character(x), '')[[1]]) # seperate the digits in the input
  sum(digits ^ 2)
}

#' Generate a sequence of happy number
#' 
#' @param x A positive integer
#' 
#' @return A list with two features:
#' \describe{
#'  \item{sequence}{A sequence of happy number starts from x. For the nth number in the 
#'  sequence, the n+1th number is the next happy number of it.}
#'  \item{isHappy}{TRUE if the sequence reaches 1, otherwise 0.}
#' }
#' 
#' @examples
#' happySequence(19)
#' happySequence(4)
#' 

happySequence <- function(x){
  if(length(x) != 1 || !is.numeric(x) || is.na(x) || !is.finite(x) || x <= 0 || x != floor(x)){ # make sure x is positive integer and not NA
    stop('Input must be a positive integer.')
  }
  if(x == 1){ # if the input in 1, just return the answer
    return(list(sequence = x, isHappy = TRUE))
  }
  values <- c(x) # sequence of values in happy sequence
  while(TRUE){
    x <- nextHappy(x)
    if(x == 1){ # if the value reaches 1, return the answer
      values <- c(values, x)
      return(list(sequence = values, isHappy = TRUE))
    }
    if(x %in% values){ # if the value repeats and will not reach 1, return the answer
      return(list(sequence = values, isHappy = FALSE))
    }
    values <- c(values, x)
  }
}
